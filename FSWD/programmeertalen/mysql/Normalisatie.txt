-- Create database
CREATE DATABASE IF NOT EXISTS courses_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;
USE courses_db;
-- Lookup: aanspreking
CREATE TABLE aanspreking (
  aanspreking_id TINYINT UNSIGNED PRIMARY KEY,
  aanspreking          VARCHAR(20) NOT NULL UNIQUE
) ENGINE=InnoDB;
-- Docent
CREATE TABLE docent (
  docent_id      INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  aanspreking_id TINYINT UNSIGNED NOT NULL,
  voornaam       VARCHAR(100) NOT NULL,
  familienaam    VARCHAR(100) NOT NULL,
  CONSTRAINT fk_docent_aanspreking
    FOREIGN KEY (aanspreking_id)
    REFERENCES aanspreking(aanspreking_id)
    ON UPDATE RESTRICT
    ON DELETE RESTRICT
) ENGINE=InnoDB;
-- Student
CREATE TABLE student (
  student_id      INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  aanspreking_id  TINYINT UNSIGNED NOT NULL,
  voornaam        VARCHAR(100) NOT NULL,
  familienaam     VARCHAR(100) NOT NULL,
  CONSTRAINT fk_student_aanspreking
    FOREIGN KEY (aanspreking_id)
    REFERENCES aanspreking(aanspreking_id)
    ON UPDATE RESTRICT
    ON DELETE RESTRICT
) ENGINE=InnoDB;
-- Vak (subject)
CREATE TABLE vak (
  vak_id        INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  naam          VARCHAR(120) NOT NULL UNIQUE,
  max_resultaat TINYINT UNSIGNED NOT NULL
) ENGINE=InnoDB;
-- Cursus (program)
CREATE TABLE cursus (
  cursus_id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  naam      VARCHAR(150) NOT NULL UNIQUE
) ENGINE=InnoDB;
-- Koppeltabel: cursus_student
CREATE TABLE cursus_student (
  id         BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  cursus_id  INT UNSIGNED NOT NULL,
  student_id INT UNSIGNED NOT NULL,
  CONSTRAINT uq_cursus_student UNIQUE (cursus_id, student_id),
  CONSTRAINT fk_cs_cursus
    FOREIGN KEY (cursus_id)
    REFERENCES cursus(cursus_id)
    ON UPDATE RESTRICT
    ON DELETE CASCADE,
  CONSTRAINT fk_cs_student
    FOREIGN KEY (student_id)
    REFERENCES student(student_id)
    ON UPDATE RESTRICT
    ON DELETE CASCADE
) ENGINE=InnoDB;
-- Resultaten student-vak
CREATE TABLE student_vak (
  id         BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  student_id INT UNSIGNED NOT NULL,
  vak_id     INT UNSIGNED NOT NULL,
  resultaat  DECIMAL(5,2) NOT NULL,
  datum      DATE NOT NULL,
  CONSTRAINT uq_student_vak_datum UNIQUE (student_id, vak_id, datum),
  CONSTRAINT fk_sv_student
    FOREIGN KEY (student_id)
    REFERENCES student(student_id)
    ON UPDATE RESTRICT
    ON DELETE CASCADE,
  CONSTRAINT fk_sv_vak
    FOREIGN KEY (vak_id)
    REFERENCES vak(vak_id)
    ON UPDATE RESTRICT
    ON DELETE RESTRICT
) ENGINE=InnoDB;
-- Koppeltabel: docent_vak
CREATE TABLE docent_vak (
  id        BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  docent_id INT UNSIGNED NOT NULL,
  vak_id    INT UNSIGNED NOT NULL,
  CONSTRAINT uq_docent_vak UNIQUE (docent_id, vak_id),
  CONSTRAINT fk_dv_docent
    FOREIGN KEY (docent_id)
    REFERENCES docent(docent_id)
    ON UPDATE RESTRICT
    ON DELETE CASCADE,
  CONSTRAINT fk_dv_vak
    FOREIGN KEY (vak_id)
    REFERENCES vak(vak_id)
    ON UPDATE RESTRICT
    ON DELETE RESTRICT
) ENGINE=InnoDB;
