CREATE DATABASE IF NOT EXISTS wedstrijduitslagen_db
	CHARACTER SET utf8mb4
 	COLLATE utf8mb4_unicode_ci;

USE wedstrijduitslagen_db;

CREATE TABLE stadion (
stadion_id TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  naam          VARCHAR(20) NOT NULL UNIQUE
);

CREATE TABLE ploegen (
	ploeg_id	TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
	naam		VARCHAR(150) NOT NULL UNIQUE,
	stadion		VARCHAR(150) NOT NULL UNIQUE
);

CREATE TABLE wedstrijden (
wedstrijd_id TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
	datum 		DATE NOT NULL;
	thuisteam 	TINYINT UNSIGNED NOT NULL,
	uitteam		TINYINT UNSIGNED NOT NULL,
	stadion		TINYINT UNSIGNED NOT NULL,
	thuisgoals 	TINYINT NOT NULL,
	uitgoals	TINYINT NOT NULL,
FOREIGN KEY (thuisteam) REFERENCES ploegen(ploeg_id)
	ON UPDATE RESTRICT ON DELETE RESTRICT,
FOREIGN KEY (uitteam) REFERENCES ploegen(ploeg_id)
	ON UPDATE RESTRICT ON DELETE RESTRICT,
FOREIGN KEY (stadion) REFERENCES stadion(stadion_id)
	ON UPDATE RESTRICT ON DELETE RESTRICT
);
