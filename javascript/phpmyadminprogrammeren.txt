SELECT co.country, ci.city
FROM country co
INNER JOIN city ci ON co.country_id =ci.country_id
;

SELECT co.country, ci.city
FROM country co
INNER JOIN city ci ON co.country_id =ci.country_id
ORDER BY co.country ASC
;

/Bij een vreemde database (gouden regel nooit inner join gebruiken) -> LEFT JOIN

SELECT co.country, ci.city
FROM country co
LEFT JOIN city ci ON co.country_id =ci.country_id
ORDER BY co.country ASC
;

// toon alle films  met de acteurs en de datum waarop de film beschikbaar werd gesteld.

SELECT f.title, a.first_name, a.last_name, f.release_year
FROM film f 
LEFT JOIN film_actor fa ON f.film_id=fa.film_id
LEFT JOIN actor a ON fa.actor_id=a.actor_id
;
