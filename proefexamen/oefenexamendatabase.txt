vraag 14 Schema in nieuwe database sql verplicht
--------------------------------------------------

CREATE TABLE klant( 
	klant_id INT(11) UNSIGNED PRIMARY KEY AUTO_INCREMENT,
 	voornaam VARCHAR(255) NOT NULL ,
 	achternaam VARCHAR(255) NOT NULL ,
 	email VARCHAR(255) NOT NULL
	);

CREATE TABLE film(
    	film_id Int(11) unsigned PRIMARY KEY AUTO_INCREMENT,
    	titel VARCHAR (255) NOT NULL
    	);        

CREATE TABLE favoriet_film(
    ff_id INT(11) UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    klant_id Int(11) UNSIGNED NOT NULL,
    film_id INT(11) UNSIGNED NOT NULL,             
    notitie VARCHAR(100) NOT NULL, 
CONSTRAINT
    FOREIGN KEY (klant_id) REFERENCES klant(klant_id),
CONSTRAINT
    FOREIGN KEY (film_id) REFERENCES film(film_id)
    ON UPDATE RESTRICT ON DELETE RESTRICT
    );
