CREATE PROCEDURE sp_totaal_uitgegeven_door_een_klant(IN p_id INT, OUT p_total DECIMAL(5,0))
BEGIN
	SELECT ROUND(sum(payment.amount),2) INTO p_total
    FROM payment
    WHERE payment.customer_id = p_id;
END //

DELIMITER ;
