SELECT LOWER(f.title) as film, f.length as lengte, length(f.title) as length_title, LEFT(f.title,1), RIGHT(f.title,1)
FROM film f;

//LEFT   Right

SELECT LOWER(f.title) as film, f.length as lengte, length(f.title) as length_title, Upper(LEFT(f.title,1)), UPPER(RIGHT(f.title,1))
FROM film f;

//upper()

SELECT LOWER(f.title) as film, f.length as lengte, length(f.title) as length_title, concat(Upper(LEFT(f.title,1)), UPPER(RIGHT(f.title,1)))
FROM film f;

//concat()  samenvoegen van data
